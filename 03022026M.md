function Mostrar-Menu {
    Clear-Host
    Write-Host "======================================" -ForegroundColor Cyan
    Write-Host "   GESTOR DE CONTRASEÑAS v1.0" -ForegroundColor Cyan
    Write-Host "======================================" -ForegroundColor Cyan
    Write-Host ""
    Write-Host "  1. Añadir nueva credencial"
    Write-Host "  2. Listar servicios guardados"
    Write-Host "  3. Recuperar contraseña"
    Write-Host "  4. Salir"
    Write-Host ""
    Write-Host "======================================" -ForegroundColor Cyan
}

function crear_credencial {

    Write-Host "`n--- NUEVA CREDENCIAL ---" -ForegroundColor Green
    $rutaFichero = "C:\Users\briof\Downloads\nombre.csv"
    
    # Pedimos los datos
    $servicio = Read-Host "Nombre del servicio (ej: Gmail, Netflix)"
    $usuario = Read-Host "Usuario o email"
    $passwordSegura = Read-Host "Contraseña" -AsSecureString

     $passwordCifrada = $passwordSegura | ConvertFrom-SecureString

        # Creamos un objeto con los datos
        $credencial = [PSCustomObject]@{
        Servicio = $servicio
        Usuario = $usuario
        PasswordCifrada = $passwordCifrada
    }
    
    # Comprobamos si el fichero existe
    if (Test-Path $rutaFichero) {
        # Si existe, añadimos la nueva línea
        $credencial | Export-Csv -Path $rutaFichero -Append -NoTypeInformation
    } else {
        # Si no existe, lo creamos
        $credencial | Export-Csv -Path $rutaFichero -NoTypeInformation
    }

}
