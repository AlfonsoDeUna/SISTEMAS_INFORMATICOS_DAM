# Obtenemos conexiones activas externas

```powershell
$conexiones = Get-NetTCPConnection | 
    Where-Object { $_.State -eq "Established" -and $_.RemoteAddress -ne "127.0.0.1" -and $_.RemoteAddress -ne "::1" }

# Creamos una tabla bonita cruzando datos
$conexiones | ForEach-Object {
    # Intentamos obtener el nombre del proceso usando el ID (PID)
    $proceso = Get-Process -Id $_.OwningProcess -ErrorAction SilentlyContinue
    
    [PSCustomObject]@{
        'IP Remota'    = $_.RemoteAddress
        'Puerto'       = $_.RemotePort
        'Programa'     = if ($proceso) { $proceso.ProcessName } else { "Desconocido (PID: $($_.OwningProcess))" }
        'ID Proceso'   = $_.OwningProcess
    }
} | Format-Table -AutoSize
```
